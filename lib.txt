local function Create(class, props)
    local object = Instance.new(class)

    for key, value in pairs(props or {}) do
        object[key] = value
    end

    return object
end

local function Tween(object, duration, properties)
    if not object or not object.Parent then
        return
    end

    local tween = TweenService:Create(
        object,
        TweenInfo.new(
            duration or 0.15,
            Enum.EasingStyle.Quint,
            Enum.EasingDirection.Out
        ),
        properties
    )

    tween:Play()

    return tween
end

local function Corner(parent, radius)
    return Create("UICorner", {
        Parent = parent,
        CornerRadius = UDim.new(0, radius or 6)
    })
end

local function Stroke(parent, color, transparency)
    return Create("UIStroke", {
        Parent = parent,
        Color = color or Library.Theme.Border,
        Transparency = transparency or 0,
        Thickness = 1
    })
end

local function Padding(parent, left, right, top, bottom)
    return Create("UIPadding", {
        Parent = parent,
        PaddingLeft = UDim.new(0, left or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or 0)
    })
end

local function AddConnection(connection)
    table.insert(Library.Connections, connection)
    return connection
end

local function MakeDraggable(handle, object)
    local dragging = false
    local dragStart
    local startPosition

    AddConnection(handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPosition = object.Position

            local connection
            connection = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false

                    if connection then
                        connection:Disconnect()
                    end
                end
            end)
        end
    end))

    AddConnection(UserInputService.InputChanged:Connect(function(input)
        if not dragging then
            return
        end

        if input.UserInputType ~= Enum.UserInputType.MouseMovement then
            return
        end

        local delta = input.Position - dragStart

        object.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end))
end

local Option = {}
Option.__index = Option

function Option:SetValue(value, silent)
    self.Value = value

    if self.Callback and not silent then
        task.spawn(self.Callback, value)
    end

    return self
end

function Option:SetText(text)
    self.Text = text

    if self.Label then
        self.Label.Text = text
    end

    return self
end

function Option:SetVisible(visible)
    if self.Container then
        self.Container.Visible = visible
    end

    return self
end

function Option:SetDisabled(disabled)
    self.Disabled = disabled

    if self.Container then
        self.Container.BackgroundTransparency = disabled and 0.35 or 0
    end

    if self.Label then
        self.Label.TextColor3 = disabled
            and Library.Theme.Muted
            or Library.Theme.Text
    end

    return self
end

function Groupbox:AddButton(...)
    local args = {...}

    local text
    local callback

    if typeof(args[1]) == "table" then
        text = args[1].Text
        callback = args[1].Func or args[1].Callback
    else
        text = args[1]
        callback = args[2]
    end

    local element = self:_CreateElement(34)

    local button = Create("TextButton", {
        Parent = element,
        BackgroundColor3 = Library.Theme.Element,
        BorderSizePixel = 0,
        Size = UDim2.new(1, 0, 1, 0),
        AutoButtonColor = false,
        Font = Enum.Font.GothamMedium,
        Text = tostring(text or "Button"),
        TextColor3 = Library.Theme.Text,
        TextSize = 13
    })

    Corner(button, 6)
    Stroke(button)

    AddConnection(button.MouseEnter:Connect(function()
        Tween(button, 0.12, {
            BackgroundColor3 = Library.Theme.Hover
        })
    end))

    AddConnection(button.MouseLeave:Connect(function()
        Tween(button, 0.12, {
            BackgroundColor3 = Library.Theme.Element
        })
    end))

    AddConnection(button.MouseButton1Click:Connect(function()
        if callback then
            task.spawn(callback)
        end
    end))

    return {
        Container = element,
        Button = button,

        SetText = function(_, value)
            button.Text = tostring(value)
        end
    }
end

function Groupbox:AddToggle(id, options)
    options = options or {}

    local option = setmetatable({}, Option)

    option.Id = id
    option.Text = options.Text or id
    option.Value = options.Default == true
    option.Default = option.Value
    option.Callback = options.Callback
    option.Disabled = false

    Library.Options[id] = option
    Library.Flags[id] = option.Value

    local element = self:_CreateElement(34)
    option.Container = element

    local button = Create("TextButton", {
        Parent = element,
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        Text = "",
        AutoButtonColor = false
    })

    local box = Create("Frame", {
        Parent = button,
        BackgroundColor3 = Library.Theme.Element,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(0, 7),
        Size = UDim2.fromOffset(20, 20)
    })

    Corner(box, 5)
    Stroke(box)

    local check = Create("TextLabel", {
        Parent = box,
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(1, 1),
        Font = Enum.Font.GothamBold,
        Text = "✓",
        TextColor3 = Library.Theme.Text,
        TextSize = 13,
        TextTransparency = option.Value and 0 or 1
    })

    local label = Create("TextLabel", {
        Parent = button,
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(29, 0),
        Size = UDim2.new(1, -29, 1, 0),
        Font = Enum.Font.Gotham,
        Text = option.Text,
        TextColor3 = Library.Theme.Text,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left
    })

    option.Label = label
    option.Button = button

    local function update(value)
        option.Value = value
        Library.Flags[id] = value

        Tween(box, 0.12, {
            BackgroundColor3 = value
                and Library.Theme.Accent
                or Library.Theme.Element
        })

        Tween(check, 0.12, {
            TextTransparency = value and 0 or 1
        })

        if options.Callback then
            task.spawn(options.Callback, value)
        end
    end

    option.SetValue = function(_, value, silent)
        option.Value = value == true
        Library.Flags[id] = option.Value

        Tween(box, 0.12, {
            BackgroundColor3 = option.Value
                and Library.Theme.Accent
                or Library.Theme.Element
        })

        Tween(check, 0.12, {
            TextTransparency = option.Value and 0 or 1
        })

        if not silent and options.Callback then
            task.spawn(options.Callback, option.Value)
        end

        return option
    end

    AddConnection(button.MouseButton1Click:Connect(function()
        if option.Disabled then
            return
        end

        update(not option.Value)
    end))

    if option.Value then
        update(true)
    end

    return option
end

function Groupbox:AddSlider(id, options)
    options = options or {}

    local min = options.Min or 0
    local max = options.Max or 100
    local default = options.Default or min
    local rounding = options.Rounding or 0

    local option = setmetatable({}, Option)

    option.Id = id
    option.Text = options.Text or id
    option.Min = min
    option.Max = max
    option.Rounding = rounding
    option.Value = default
    option.Default = default
    option.Callback = options.Callback

    Library.Options[id] = option
    Library.Flags[id] = default

    local element = self:_CreateElement(52)
    option.Container = element

    local label = Create("TextLabel", {
        Parent = element,
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(0, 0),
        Size = UDim2.new(1, -55, 0, 22),
        Font = Enum.Font.Gotham,
        Text = option.Text,
        TextColor3 = Library.Theme.Text,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left
    })

    local valueLabel = Create("TextLabel", {
        Parent = element,
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, 0, 0, 0),
        Size = UDim2.fromOffset(55, 22),
        Font = Enum.Font.GothamMedium,
        Text = tostring(default),
        TextColor3 = Library.Theme.Accent,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Right
    })

    local bar = Create("Frame", {
        Parent = element,
        BackgroundColor3 = Library.Theme.Element,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(0, 29),
        Size = UDim2.new(1, 0, 0, 6)
    })

    Corner(bar, 3)

    local fill = Create("Frame", {
        Parent = bar,
        BackgroundColor3 = Library.Theme.Accent,
        BorderSizePixel = 0,
        Size = UDim2.new(
            math.clamp((default - min) / (max - min), 0, 1),
            0,
            1,
            0
        )
    })

    Corner(fill, 3)

    local hitbox = Create("TextButton", {
        Parent = element,
        BackgroundTransparency = 1,
        Position = UDim2.fromOffset(0, 23),
        Size = UDim2.new(1, 0, 0, 18),
        Text = ""
    })

    option.Label = label

    local dragging = false

    local function roundValue(value)
        local multiplier = 10 ^ rounding
        return math.floor(value * multiplier + 0.5) / multiplier
    end

    local function setValue(value, silent)
        value = math.clamp(value, min, max)
        value = roundValue(value)

        option.Value = value
        Library.Flags[id] = value

        local percentage = (value - min) / (max - min)

        fill.Size = UDim2.new(percentage, 0, 1, 0)
        valueLabel.Text = tostring(value)

        if not silent and options.Callback then
            task.spawn(options.Callback, value)
        end
    end

    option.SetValue = function(_, value, silent)
        setValue(value, silent)
        return option
    end

    AddConnection(hitbox.MouseButton1Down:Connect(function()
        dragging = true
    end))

    AddConnection(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end))

    AddConnection(UserInputService.InputChanged:Connect(function(input)
        if not dragging then
            return
        end

        if input.UserInputType ~= Enum.UserInputType.MouseMovement then
            return
        end

        local percentage = math.clamp(
            (input.Position.X - bar.AbsolutePosition.X)
                / bar.AbsoluteSize.X,
            0,
            1
        )

        setValue(min + (max - min) * percentage)
    end))

    setValue(default, true)

    return option
end

function Groupbox:AddDropdown(id, options)
    options = options or {}

    local values = options.Values or {}
    local multi = options.Multi == true

    local option = setmetatable({}, Option)

    option.Id = id
    option.Text = options.Text or id
    option.Values = values
    option.Multi = multi
    option.Callback = options.Callback

    if multi then
        option.Value = options.Default or {}
    else
        option.Value = options.Default or values[1]
    end

    Library.Options[id] = option
    Library.Flags[id] = option.Value

    local element = self:_CreateElement(36)
    option.Container = element

    local label = Create("TextLabel", {
        Parent = element,
        BackgroundTransparency = 1,
        Size = UDim2.new(0.45, 0, 1, 0),
        Font = Enum.Font.Gotham,
        Text = option.Text,
        TextColor3 = Library.Theme.Text,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left
    })

    local dropdown = Create("TextButton", {
        Parent = element,
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, 0, 0, 0),
        Size = UDim2.new(0.52, 0, 1, 0),
        BackgroundColor3 = Library.Theme.Element,
        BorderSizePixel = 0,
        AutoButtonColor = false,
        Font = Enum.Font.Gotham,
        TextColor3 = Library.Theme.Text,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left,
        Text = ""
    })

    Corner(dropdown, 5)
    Stroke(dropdown)
    Padding(dropdown, 10, 8)

    local arrow = Create("TextLabel", {
        Parent = dropdown,
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, 0, 0.5, 0),
        Size = UDim2.fromOffset(20, 20),
        BackgroundTransparency = 1,
        Font = Enum.Font.GothamBold,
        Text = "⌄",
        TextColor3 = Library.Theme.SubText,
        TextSize = 14
    })

    local list = Create("ScrollingFrame", {
        Parent = element,
        Visible = false,
        Position = UDim2.new(0.48, 0, 1, 4),
        Size = UDim2.new(0.52, 0, 0, 0),
        BackgroundColor3 = Library.Theme.Secondary or Library.Theme.Sidebar,
        BorderSizePixel = 0,
        ScrollBarThickness = 2,
        CanvasSize = UDim2.new(),
        ZIndex = 20
    })

    Corner(list, 5)
    Stroke(list)

    Create("UIListLayout", {
        Parent = list,
        SortOrder = Enum.SortOrder.LayoutOrder,
        Padding = UDim.new(0, 2)
    })

    local function getText()
        if multi then
            local selected = {}

            for value, enabled in pairs(option.Value) do
                if enabled then
                    table.insert(selected, tostring(value))
                end
            end

            if #selected == 0 then
                return "None"
            end

            return table.concat(selected, ", ")
        end

        return tostring(option.Value or "None")
    end

    local function refreshText()
        dropdown.Text = getText()
    end

    local function rebuild()
        for _, child in ipairs(list:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end

        local count = #values

        for index, value in ipairs(values) do
            local button = Create("TextButton", {
                Parent = list,
                BackgroundColor3 = Library.Theme.Element,
                BorderSizePixel = 0,
                Size = UDim2.new(1, -4, 0, 28),
                AutoButtonColor = false,
                Font = Enum.Font.Gotham,
                Text = tostring(value),
                TextColor3 = Library.Theme.Text,
                TextSize = 12,
                ZIndex = 21
            })

            Corner(button, 4)

            local selected

            if multi then
                selected = option.Value[value] == true
            else
                selected = option.Value == value
            end

            if selected then
                button.BackgroundColor3 = Library.Theme.Accent
            end

            AddConnection(button.MouseButton1Click:Connect(function()
                if multi then
                    option.Value[value] = not option.Value[value]
                    Library.Flags[id] = option.Value
                else
                    option.Value = value
                    Library.Flags[id] = value
                    list.Visible = false
                end

                refreshText()
                rebuild()

                if options.Callback then
                    task.spawn(options.Callback, option.Value)
                end
            end))
        end

        list.CanvasSize = UDim2.fromOffset(
            0,
            count * 30 + 4
        )

        list.Size = UDim2.new(
            0.52,
            0,
            0,
            math.min(count * 30 + 4, 160)
        )
    end

    AddConnection(dropdown.MouseButton1Click:Connect(function()
        list.Visible = not list.Visible
        arrow.Text = list.Visible and "⌃" or "⌄"
    end))

    option.SetValue = function(_, value, silent)
        option.Value = value
        Library.Flags[id] = value

        refreshText()
        rebuild()

        if not silent and options.Callback then
            task.spawn(options.Callback, value)
        end

        return option
    end

    option.SetValues = function(_, newValues)
        values = newValues or {}
        option.Values = values
        rebuild()
        refreshText()
        return option
    end

    refreshText()
    rebuild()

    return option
end

    AddConnection(UserInputService.InputBegan:Connect(function(input, processed)
        if not listening or processed then
            return
        end

        if input.UserInputType == Enum.UserInputType.Keyboard then
            option.Value = input.KeyCode.Name
            Library.Flags[id] = option.Value

            button.Text = option.Value
            listening = false

            if options.Callback then
                task.spawn(options.Callback, option.Value)
            end
        end
    end))

    option.SetValue = function(_, value, silent)
        option.Value = value
        Library.Flags[id] = value
        button.Text = tostring(value)

        if not silent and options.Callback then
            task.spawn(options.Callback, value)
        end

        return option
    end

    return option
end

        current.Content.Visible = selected

                and Library.Theme.Text
                or Library.Theme.SubText,

                and Library.Theme.Element
                or Library.Theme.Sidebar
        })

            BackgroundTransparency = selected and 0 or 1
        })
    end

Library.Configs = {}

function Library:SetConfigFolder(name)
    self.ConfigFolder = tostring(name)
end

local function serialize(value)
    local valueType = typeof(value)

    if valueType == "Color3" then
        return {
            __type = "Color3",
            R = value.R,
            G = value.G,
            B = value.B
        }
    end

    if valueType == "EnumItem" then
        return {
            __type = "EnumItem",
            EnumType = tostring(value.EnumType),
            Name = value.Name
        }
    end

    if type(value) == "table" then
        local result = {}

        for key, item in pairs(value) do
            result[key] = serialize(item)
        end

        return result
    end

    if type(value) == "string"
        or type(value) == "number"
        or type(value) == "boolean"
        or value == nil then

        return value
    end

    return tostring(value)
end

local function deserialize(value)
    if type(value) ~= "table" then
        return value
    end

    if value.__type == "Color3" then
        return Color3.new(
            value.R,
            value.G,
            value.B
        )
    end

    local result = {}

    for key, item in pairs(value) do
        result[key] = deserialize(item)
    end

    return result
end

function Library:GetConfigData()
    local data = {}

    for id, option in pairs(self.Options) do
        data[id] = serialize(option.Value)
    end

    return data
end

function Library:SaveConfig(name)
    name = tostring(name)

    self.Configs[name] = self:GetConfigData()

    return true
end

function Library:LoadConfig(name)
    name = tostring(name)

    local data = self.Configs[name]

    if not data then
        return false
    end

    for id, value in pairs(data) do
        local option = self.Options[id]

        if option then
            option:SetValue(deserialize(value), true)
        end
    end

    return true
end

function Library:DeleteConfig(name)
    name = tostring(name)

    if self.Configs[name] then
        self.Configs[name] = nil
        return true
    end

    return false
end

function Library:GetConfigs()
    local result = {}

    for name in pairs(self.Configs) do
        table.insert(result, name)
    end

    table.sort(result)

    return result
end

--==================================================
-- UNLOAD
--==================================================

function Library:Unload()
    if self.Unloaded then
        return
    end

    self.Unloaded = true

    for _, connection in ipairs(self.Connections) do
        pcall(function()
            connection:Disconnect()
        end)
    end

    table.clear(self.Connections)
    table.clear(self.Options)
    table.clear(self.Flags)

    if self.Gui then
        self.Gui:Destroy()
        self.Gui = nil
    end

    self.Root = nil
    self.Tabs = {}
end

return Library
